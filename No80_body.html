$Kit = "$HOME\Desktop\midashi_kit"
$Need = @("header.ps1", "settings.csv", "README.txt", "check.ps1")

$ok = $true
foreach ($n in $Need) {
    $p = Join-Path $Kit $n
    if (-not (Test-Path $p)) { "不足: $n"; $ok = $false; continue }
    $f = Get-Item $p
    if ($f.Length -eq 0) { "中身が空: $n"; $ok = $false; continue }
    $head = [byte[]](Get-Content $p -Encoding Byte -TotalCount 3)
    $bom = ($head.Length -eq 3 -and $head[0] -eq 0xEF -and $head[1] -eq 0xBB -and $head[2] -eq 0xBF)
    if (-not $bom) { "印なし（相手の画面で日本語が化けます）: $n"; $ok = $false; continue }
    "OK  {0,-14} {1,6} bytes" -f $n, $f.Length
}

$extra = Get-ChildItem $Kit -File | Where-Object { $Need -notcontains $_.Name }
foreach ($e in $extra) { "余分: $($e.Name)"; $ok = $false }

if ($ok) { "渡せます（$($Need.Count)点）" } else { "直すところがあります" }

見ているのは4つです。

  • 決めた4点が揃っているか
  • 中身が空のファイルが混ざっていないか
  • 文字コードの印が入っているか（4つのうち、目で見えないのはここだけ）
  • 決めた4点以外が入っていないか
